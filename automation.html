04 / AUTOMATION CODE

Playwright spec

JAVASCRIPT

A representative snippet from the repository's runnable browser tests.

import { test, expect } from '@playwright/test';

test('portfolio links to QA tools', async ({ page }) => {
  await page.goto('/');
  await expect(page).toHaveTitle(/QA Engineer Portfolio/);
  await expect(page.getByRole('link', {
    name: /OPEN INTERACTIVE QA LAB/
  })).toHaveAttribute('href', 'qa-lab.html');
});

test('page has no horizontal overflow', async ({ page }) => {
  await page.setViewportSize({ width: 375, height: 812 });
  await page.goto('/');
  const width = await page.evaluate(() =>
    document.documentElement.scrollWidth
  );
  expect(width).toBeLessThanOrEqual(375);
});
